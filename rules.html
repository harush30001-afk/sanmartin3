rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signed() { return request.auth != null && request.auth.token.email_verified == true; }
    function isOwner() { return signed() && request.auth.token.email == 'harush30001@gmail.com'; }
    function member() { return signed() && exists(/databases/$(database)/documents/team/$(request.auth.token.email)); }
    // הפרויקטים שהמשתמש מורשה אליהם. משתמש ישן בלי שדה projects מקבל רק את סאן מרטין 3
    function projs() { return get(/databases/$(database)/documents/team/$(request.auth.token.email)).data.get('projects', ['sm3']); }
    function hasProj(p) { return isOwner() || (member() && p in projs()); }
    // גרסה 632: בוט בניית המודלים (משימה מתוזמנת של Claude). רק תור המודלים ותמונות
    function isBot() { return request.auth != null && request.auth.token.email == 'harush30001+yhbot@gmail.com'; }

    // הנהלה: מנהל פרויקט, מנהל פרויקט בכיר, מנהל אזור, מהנדס ראשי — יכולים לשנות הרשאות מפקח בפרויקט שלהם
    function myJr() { return get(/databases/$(database)/documents/team/$(request.auth.token.email)).data.get('jr', ''); }
    function mgr() { return member() && myJr() in ['pm', 'spm', 'area', 'chief']; }
    function supMine(d) { return d.get('jr', '') == 'sup' && d.get('projects', ['sm3']).hasAny(projs()); }
    function invOk(email) {
      let d = request.resource.data;
      let i = get(/databases/$(database)/documents/sign/$(d.inv)).data;
      return signed() && request.auth.token.email == email && d.email == email
        && i.get('t', '') == 'inv' && i.get('st', '') == 'p' && i.get('exp', 0) > request.time.toMillis()
        && d.get('jr', '') == i.get('jr', '') && d.role == i.role && d.projects == i.prj
        && d.keys().hasOnly(['email', 'name', 'ph', 'jr', 'role', 'projects', 'inv', 'by']);
    }
    match /team/{email} {
      allow read: if signed() && (request.auth.token.email == email || isOwner() || (mgr() && supMine(resource.data)));
      allow write: if isOwner();
      // גרסה 376: הצטרפות רק באישור הבעלים — המוזמן לא יוצר לעצמו הרשאה
      allow update: if mgr() && supMine(resource.data) && request.resource.data.get('jr', '') == 'sup'
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['cols']);
    }
    // חתימה מרחוק: מי שיש לו את הקישור (מזהה אקראי) יכול לקרוא את הבקשה ולחתום פעם אחת בלבד
    // גרסה 377: הזמנות (t == 'inv') — רק הבעלים יוצר, משנה ומוחק. חבר צוות לא נוגע בהן ולא יכול להתחזות להזמנה
    function isInv(d) { return d.get('t', '') == 'inv'; }
    match /sign/{tok} {
      allow get: if true;
      allow list: if isOwner() || member() || isBot();
      allow create: if isOwner() || (member() && !isInv(request.resource.data)) || (isBot() && request.resource.data.get('k', '') == 'mdl');
      allow delete: if isOwner() || (member() && !isInv(resource.data));
      allow update: if isOwner()
        || (isBot() && resource.data.get('k', '') == 'mq' && request.resource.data.k == 'mq')
        || (member() && !isInv(resource.data) && !isInv(request.resource.data))
        // קבלן מהקישור: רק תשובות (טופל / שאלה) על הליקויים שנשלחו אליו. שום שדה אחר
        || (
          resource.data.get('k', '') == 'df' && resource.data.st == 'v' &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['r', 'rt']) &&
          request.resource.data.r is map && request.resource.data.r.size() <= 60
        )
        // גרסה 591: דייר אחרי מסירה (k == 'tw'): רק מוסיף דיווחים לרשימה q (עד 30), שום שדה אחר
        || (
          resource.data.get('k', '') == 'tw' && resource.data.st == 'v' &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['q', 'qt']) &&
          request.resource.data.q is list && request.resource.data.q.size() <= 30 &&
          request.resource.data.q.size() >= resource.data.get('q', []).size()
        )
        || (
          resource.data.st == 'p' && request.resource.data.st == 's' &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['st', 'sg', 'nm', 'ts']) &&
          request.resource.data.sg is string && request.resource.data.sg.size() < 400000 &&
          request.resource.data.nm is string && request.resource.data.nm.size() < 120 &&
          // מי שמגיש בקשת הצטרפות — רק בשם המייל שאיתו נכנס בפועל, ורק לפני שפג התוקף
          (!isInv(resource.data) || (signed() && request.resource.data.nm == request.auth.token.email.lower() && resource.data.get('exp', 0) > request.time.toMillis()))
        );
    }
    match /backups/{d} {
      allow read, write: if isOwner();
    }
    // הגדרת פרויקט: רק מי שמורשה קורא, רק הבעלים כותב
    match /projects/{pid} {
      allow read: if hasProj(pid);
      allow write: if isOwner();
      // סיכום חי של הפרויקט (למנהלי אזור ולמהנדס הראשי) — כל חבר צוות בפרויקט מעדכן רק את שדה הסיכום
      allow update: if hasProj(pid) && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['sum']);
    }
    // תפקידים: editor = צוות האתר, store = מחסנאי, viewer = מפקח (צפייה), contractor = קבלן (ישן). מנהל = רק המייל של הבעלים
    function role() { return get(/databases/$(database)/documents/team/$(request.auth.token.email)).data.get('role', 'editor'); }
    function canEd(p) { return isOwner() || (hasProj(p) && role() == 'editor'); }
    function canTask(p) { return isOwner() || (hasProj(p) && role() in ['editor', 'contractor']); }
    function canMat(p) { return isOwner() || (hasProj(p) && role() in ['editor', 'store']); }
    function canPh(p) { return isOwner() || (hasProj(p) && role() in ['editor', 'contractor', 'store']); }
    function shape() { return request.resource.data.keys().hasOnly(['j', 'd', 'del', 't', 'by']); }
    function okw(p, c) {
      return shape() && (
        (c in ['cl', 'lists', 'appts', 'meta', 'log', 'dmg', 'x'] && canEd(p)) ||
        (c == 'tasks' && canTask(p)) ||
        (c == 'mat' && canMat(p)) ||
        (c == 'photos' && canPh(p) && (!('d' in request.resource.data) || request.resource.data.d.size() < 900000))
      );
    }
    // נתוני פרויקט חדש
    match /projects/{pid}/{pc}/{pd} {
      allow read: if (hasProj(pid) && pc in ['cl','lists','tasks','log','appts','meta','photos','mat','dmg','x']) || (isBot() && pc == 'photos');
      allow create: if okw(pid, pc);
      // גרסה 422: פרוטוקול מסירה חתום (x/hoS_*) — רק הבעלים משנה או מוחק. אחרי שהבעלים פתח אותו אפשר לחתום מחדש
      allow update: if okw(pid, pc) && (!(pc == 'x' && pd.matches('hoS_.*')) || isOwner() || resource.data.get('del', 0) == 1);
      allow delete: if isOwner() && pc in ['cl','lists','tasks','log','appts','meta','photos','mat','dmg','x'];
    }
    // סאן מרטין 3 (הנתונים הקיימים, בלי שינוי מיקום)
    match /{col}/{d} {
      allow read: if (hasProj('sm3') && col in ['cl','lists','tasks','log','appts','meta','photos','mat','dmg','x']) || (isBot() && col == 'photos');
      allow create: if okw('sm3', col);
      allow update: if okw('sm3', col) && (!(col == 'x' && d.matches('hoS_.*')) || isOwner() || resource.data.get('del', 0) == 1);
      allow delete: if isOwner() && col in ['cl','lists','tasks','log','appts','meta','photos','mat','dmg','x'];
    }
  }
}
