rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signed() { return request.auth != null && request.auth.token.email_verified == true; }
    function isOwner() { return signed() && request.auth.token.email == 'harush30001@gmail.com'; }
    function member() { return signed() && exists(/databases/$(database)/documents/team/$(request.auth.token.email)); }
    // הפרויקטים שהמשתמש מורשה אליהם. משתמש ישן בלי שדה projects מקבל רק את סאן מרטין 3
    function projs() { return get(/databases/$(database)/documents/team/$(request.auth.token.email)).data.get('projects', ['sm3']); }
    function hasProj(p) { return isOwner() || (member() && p in projs()); }
    // גרסה 632: בוט בניית המודלים (משימה מתוזמנת של Claude). רק תור המודלים ותמונות
    function isBot() { return request.auth != null && request.auth.token.email == 'harush30001+yhbot@gmail.com'; }

    // הנהלה: מנהל פרויקט, מנהל פרויקט בכיר, מנהל אזור, מהנדס ראשי — יכולים לשנות הרשאות מפקח בפרויקט שלהם
    function myJr() { return get(/databases/$(database)/documents/team/$(request.auth.token.email)).data.get('jr', ''); }
    function mgr() { return member() && myJr() in ['pm', 'spm', 'area', 'chief']; }
    function supMine(d) { return d.get('jr', '') == 'sup' && d.get('projects', ['sm3']).hasAny(projs()); }
    function invOk(email) {
      let d = request.resource.data;
      let i = get(/databases/$(database)/documents/sign/$(d.inv)).data;
      return signed() && request.auth.token.email == email && d.email == email
        && i.get('t', '') == 'inv' && i.get('st', '') == 'p' && i.get('exp', 0) > request.time.toMillis()
        && d.get('jr', '') == i.get('jr', '') && d.role == i.role && d.projects == i.prj
        && d.keys().hasOnly(['email', 'name', 'ph', 'jr', 'role', 'projects', 'inv', 'by']);
    }
    match /team/{email} {
      allow read: if signed() && (request.auth.token.email == email || isOwner() || (mgr() && supMine(resource.data)));
      allow write: if isOwner();
      // גרסה 376: הצטרפות רק באישור הבעלים — המוזמן לא יוצר לעצמו הרשאה
      allow update: if mgr() && supMine(resource.data) && request.resource.data.get('jr', '') == 'sup'
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['cols']);
    }
    // חתימה מרחוק: מי שיש לו את הקישור (מזהה אקראי) יכול לקרוא את הבקשה ולחתום פעם אחת בלבד
    // גרסה 377: הזמנות (t == 'inv') — רק הבעלים יוצר, משנה ומוחק. חבר צוות לא נוגע בהן ולא יכול להתחזות להזמנה
    function isInv(d) { return d.get('t', '') == 'inv'; }
    // גרסה 858 (בדיקת אבטחה, ממצא 1): כל מסמך sign שייך לפרויקט (prj). מסמך ישן בלי prj שייך לסאן מרטין 3
    function sPrj(d) { return d.get('prj', 'sm3'); }
    function isSgn(d) { return d.get('st', '') == 's'; }
    // צוות הפרויקט של המסמך
    function sTeam(d) { return member() && hasProj(sPrj(d)); }
    // גרסה 858 (יהב: "חתימה שנחתמה = ברזל"): מסמך חתום (st == 's') — רק הבעלים משנה או מוחק. לכל השאר — ברזל
    // (גם מנהל פרויקט). הזמנות (t == 'inv') ממילא רק של הבעלים
    function frozen(d) { return isSgn(d) && !isInv(d); }
    // מי רואה חתימות ופרטי דיירים: עוזר מנהל עבודה ומעלה, ובעלים. לא מחסנאי, מפקח, לוגיסטיקה או בטיחות
    function piiJr() { return ['afm', 'fm', 'fms', 'fmf', 'eng', 'pm', 'spm', 'area', 'chief']; }
    function seePii(p) { return isOwner() || (member() && p in projs() && myJr() in piiJr()); }
    // מסמכי sign חתומים שאין בהם פרטי דייר: תשובת ספק (spr), הרשמת קבלן (reg), קישור לקבלן (ctl)
    function noPii(d) { return d.get('k', '') in ['spr', 'reg', 'ctl']; }
    // ממצא 2: תשובה אחת של קבלן מהקישור — "טופל" (d) או "יש בעיה" (q), הערה ותמונה אחת. בלי שם דייר, חתימת דייר או "אומת בת.ז."
    function dfRepOk(v) {
      return v is map && v.keys().hasOnly(['st', 'n', 'af', 'by', 't'])
        && v.st in ['d', 'q', 'm']
        && v.get('n', '') is string && v.get('n', '').size() <= 300
        // 871: "זה לא שלי · בקש העברה" (m) — רק עם הערה
        && (v.st != 'm' || v.get('n', '').size() > 0)
        && v.get('by', '') is string && v.get('by', '').size() <= 60
        && v.get('t', 0) is number
        && v.get('af', []) is list && v.get('af', []).size() <= 1
        && (v.get('af', []).size() == 0 || (v.af[0] is string && v.af[0].size() < 120000));
    }
    // הקבלן משנה לכל היותר ליקוי אחד בכל כתיבה, לא מוחק אף ליקוי מהמפה,
    // ודחייה של הצוות (st == 'x') מתחלפת רק בדיווח חדש ממנה (תיקון חוזר), לא בדיווח ישן ולא ב-x מזויף
    function dfOne(o, n) {
      let ch = n.diff(o);
      let nv = n.values().removeAll(o.values());
      let gone = o.values().removeAll(n.values());
      return ch.removedKeys().size() == 0 && ch.affectedKeys().size() <= 1
        && nv.size() == ch.affectedKeys().size()
        && (nv.size() == 0 || (dfRepOk(nv[0])
          && (gone.size() == 0 || gone[0].get('st', '') != 'x' || nv[0].t > gone[0].get('t', 0))));
    }
    // רשימה שאפשר רק להוסיף לה: כל מה שהיה בה נשאר
    function keepsAll(oldL, newL) { return oldL.removeAll(newL).size() == 0; }
    match /sign/{tok} {
      // קריאה רק לפי מזהה (מי שיש לו את הקישור). אין רשימה של טוקנים — רק הבעלים (אישור הצטרפות) והבוט (תור המודלים בלבד)
      // לפני חתימה: כל מי שיש לו את הקישור. אחרי חתימה: חתימה ופרטי דייר רק לעוזר מנהל עבודה ומעלה בפרויקט.
      // המוזמן רואה את הבקשה שהגיש בעצמו (nm == המייל שלו)
      allow get: if !isSgn(resource.data) || noPii(resource.data) || seePii(sPrj(resource.data))
        || (isInv(resource.data) && signed() && resource.data.get('nm', '') == request.auth.token.email.lower());
      allow list: if isOwner() || (isBot() && resource.data.get('k', '') == 'mq');
      allow create: if isOwner()
        || (sTeam(request.resource.data) && !isInv(request.resource.data) && !isSgn(request.resource.data))
        // יומן שגיאות AI (aie) — כל חבר צוות, בלי פרויקט, רק יצירה
        || (member() && request.resource.data.get('k', '') == 'aie' && !('prj' in request.resource.data)
            && request.resource.data.keys().hasOnly(['k', 'e', 'lg', 'v', 'ts', 'ua', 'kt', 'bt', 'bs', 'u']))
        || (isBot() && request.resource.data.get('k', '') == 'mdl');
      allow delete: if isOwner()
        || (!frozen(resource.data) && sTeam(resource.data) && !isInv(resource.data));
      allow update: if isOwner()
        || (isBot() && resource.data.get('k', '') == 'mq' && request.resource.data.k == 'mq')
        // צוות הפרויקט בלבד, בלי להעביר לפרויקט אחר. מסמך חתום — אף אחד. איש צוות לא מסמן "נחתם" בעצמו
        || (sTeam(resource.data) && sPrj(request.resource.data) == sPrj(resource.data)
            && !isInv(resource.data) && !isInv(request.resource.data)
            && !isSgn(resource.data) && !isSgn(request.resource.data))
        // קבלן מהקישור: רק תשובות (טופל / שאלה) על הליקויים שנשלחו אליו. שום שדה אחר, ובלי חתימה או שם של דייר
        // ("נחתם ע״י הדייר" — dn[].sgn / sgd / sgi — כותב רק הצוות, במסלול הצוות שלמעלה. הקבלן רק קורא)
        || (
          resource.data.get('k', '') == 'df' && resource.data.st == 'v' &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['r', 'rt']) &&
          request.resource.data.get('rt', 0) is number &&
          request.resource.data.r is map && request.resource.data.r.size() <= 60 &&
          dfOne(resource.data.get('r', {}), request.resource.data.r)
        )
        // גרסה 708: נציג הוועד בסבב חתימות (k == 'pr'): פעם אחת — תשובות ✓/✗ (r), חתימה, שם. שום שדה אחר
        || (
          resource.data.get('k', '') == 'pr' && resource.data.st == 'v' && request.resource.data.st == 's' &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['r', 'st', 'sg', 'nm', 'ts']) &&
          request.resource.data.r is map && request.resource.data.r.size() <= 60 &&
          request.resource.data.sg is string && request.resource.data.sg.size() < 400000 &&
          request.resource.data.nm is string && request.resource.data.nm.size() < 120
        )
        // גרסה 591: דייר אחרי מסירה (k == 'tw'): רק מוסיף דיווחים לרשימה q (עד 30), לא משנה ולא מוחק קודמים. שום שדה אחר
        // 871: רק אחרי מסירה חתומה + קבלת מפתח (ok == 1, נכתב ע"י הצוות)
        || (
          resource.data.get('k', '') == 'tw' && resource.data.st == 'v' && resource.data.get('ok', 0) == 1 &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['q', 'qt']) &&
          request.resource.data.q is list && request.resource.data.q.size() <= 30 &&
          keepsAll(resource.data.get('q', []), request.resource.data.q)
        )
        // גרסה 855: ממונה בטיחות מהקישור (k == 'sfl'): מוסיף ממצאים (q, עד 120) ומאשר / מחזיר תיקונים (ap). שום שדה אחר
        || (
          resource.data.get('k', '') == 'sfl' && resource.data.st == 'p' &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['q', 'qt', 'ap']) &&
          request.resource.data.get('q', []) is list && request.resource.data.get('q', []).size() <= 120 &&
          keepsAll(resource.data.get('q', []), request.resource.data.get('q', [])) &&
          request.resource.data.get('ap', {}) is map && request.resource.data.get('ap', {}).size() <= 500
        )
        // גרסה 857: קישור לקבלן (k == 'ctl'): בטעינה הראשונה הקבלן רושם פעם אחת רק את op (מספר), לפני שענה
        || (
          resource.data.get('k', '') == 'ctl' && resource.data.st == 'p' && !('op' in resource.data) &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['op']) &&
          request.resource.data.op is number
        )
        // חתימה רגילה מהקישור (דייר, ספק, הרשמת קבלן, אישור הגעה): פעם אחת בלבד, p → s, רק חתימה, שם וזמן
        || (
          resource.data.st == 'p' && request.resource.data.st == 's' &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['st', 'sg', 'nm', 'ts']) &&
          request.resource.data.sg is string && request.resource.data.sg.size() < 400000 &&
          request.resource.data.nm is string && request.resource.data.nm.size() < 120 &&
          // מי שמגיש בקשת הצטרפות — רק בשם המייל שאיתו נכנס בפועל, ורק לפני שפג התוקף
          (!isInv(resource.data) || (signed() && request.resource.data.nm == request.auth.token.email.lower() && resource.data.get('exp', 0) > request.time.toMillis()))
        );
    }
    // גרסה 858: פרטי דיירים וחתימות — אוסף נפרד, מחוץ ל-JSON של הדירה (j), כדי שאפשר יהיה להגביל לפי תפקיד.
    //   pii/{pid}/tn/{aptKey}  — שם, טלפון, ת.ז. של הדיירים בדירה (משנים כשצריך)
    //   pii/{pid}/sg/{id}      — תמונת חתימה (נכתבת פעם אחת, לא משתנה ולא נמחקת)
    match /pii/{pid}/tn/{d} {
      allow read, create, update: if seePii(pid);
      allow delete: if isOwner();
    }
    // 871: חתימת דייר על "תוקן" מהקישור שלו — מסמך אחד לכל ליקוי (sg_tw_<טוקן>_<ליקוי>), רק כשהקבלן סימן תוקן (qs[ליקוי] == 'w').
    // נכתבת פעם אחת: אין update / delete לאף אחד, ומסמך קיים לא נכתב שוב (create בלבד)
    function twSg(pid, id) {
      let d = request.resource.data;
      return d.keys().hasOnly(['d', 't', 'nm', 'by', 'q'])
        && d.d is string && d.d.size() < 200000 && d.t is number
        && d.nm is string && d.nm.size() > 0 && d.nm.size() < 120
        && d.q is string && d.q.matches('[0-9]{1,16}')
        && d.by is string && d.by.matches('tw:[a-z0-9]{10,40}')
        && id == 'sg_tw_' + d.by.split(':')[1] + '_' + d.q
        && twSgTok(get(/databases/$(database)/documents/sign/$(d.by.split(':')[1])).data, pid, d.q);
    }
    function twTokOk(s, p) { return s.get('k', '') == 'tw' && s.get('st', '') == 'v' && s.get('ok', 0) == 1 && s.get('prj', 'sm3') == p; }
    function twSgTok(s, p, q) { return twTokOk(s, p) && s.get('qs', {}).get(q, '') == 'w'; }
    match /pii/{pid}/sg/{d} {
      allow read: if seePii(pid);
      allow create: if (seePii(pid) && request.resource.data.get('d', '') is string && request.resource.data.get('d', '').size() < 400000)
        || twSg(pid, d);
      allow update, delete: if false;
    }
    match /backups/{d} {
      allow read, write: if isOwner();
    }
    // הגדרת פרויקט: רק מי שמורשה קורא, רק הבעלים כותב
    match /projects/{pid} {
      allow read: if hasProj(pid);
      allow write: if isOwner();
      // סיכום חי של הפרויקט (למנהלי אזור ולמהנדס הראשי) — כל חבר צוות בפרויקט מעדכן רק את שדה הסיכום
      allow update: if hasProj(pid) && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['sum']);
    }
    // תפקידים: editor = צוות האתר, store = מחסנאי, viewer = מפקח (צפייה), contractor = קבלן (ישן). מנהל = רק המייל של הבעלים
    function role() { return get(/databases/$(database)/documents/team/$(request.auth.token.email)).data.get('role', 'editor'); }
    function canEd(p) { return isOwner() || (hasProj(p) && role() == 'editor'); }
    function canTask(p) { return isOwner() || (hasProj(p) && role() in ['editor', 'contractor']); }
    function canMat(p) { return isOwner() || (hasProj(p) && role() in ['editor', 'store']); }
    function canPh(p) { return isOwner() || (hasProj(p) && role() in ['editor', 'contractor', 'store']); }
    function shape() { return request.resource.data.keys().hasOnly(['j', 'd', 'del', 't', 'by']); }
    function okw(p, c) {
      return shape() && (
        (c in ['cl', 'lists', 'appts', 'meta', 'log', 'dmg', 'x'] && canEd(p)) ||
        (c == 'tasks' && canTask(p)) ||
        (c == 'mat' && canMat(p)) ||
        (c == 'photos' && canPh(p) && (!('d' in request.resource.data) || request.resource.data.d.size() < 900000))
      );
    }
    // גרסה 869 (860.5, יהב 26.11): תמונת "טופל" מקישור הקבלן — נכתבת ישר ל-photos/pf* (מסמך חדש בלבד).
    // רק d (התמונה, עד 120KB), t ו-by = 'df:' + הטוקן של קישור ליקויים פעיל (sign/tok: k == 'df', st == 'v', ph == 1, אותו פרויקט)
    function dfPh(id, p) {
      let d = request.resource.data;
      return id.matches('pf[0-9a-z]{6,40}') && d.keys().hasOnly(['d', 't', 'by'])
        && d.d is string && d.d.size() < 120000 && d.t is number
        && d.by is string && d.by.matches('df:[A-Za-z0-9_-]{6,40}')
        && dfTokOk(get(/databases/$(database)/documents/sign/$(d.by.split(':')[1])).data, p);
    }
    function dfTokOk(s, p) { return s.get('k', '') == 'df' && s.get('st', '') == 'v' && s.get('ph', 0) == 1 && s.get('prj', 'sm3') == p; }
    // 871: תמונת ליקוי מהדייר (1–2) — photos/pt* (מסמך חדש בלבד), רק d (עד 120KB), t, by = 'tw:' + טוקן של קישור דייר פעיל
    // אחרי מסירה חתומה + מפתח (sign/tok: k == 'tw', st == 'v', ok == 1, אותו פרויקט)
    function twPh(id, p) {
      let d = request.resource.data;
      return id.matches('pt[0-9a-z]{6,40}') && d.keys().hasOnly(['d', 't', 'by'])
        && d.d is string && d.d.size() < 120000 && d.t is number
        && d.by is string && d.by.matches('tw:[a-z0-9]{10,40}')
        && twTokOk(get(/databases/$(database)/documents/sign/$(d.by.split(':')[1])).data, p);
    }
    // נתוני פרויקט חדש
    match /projects/{pid}/{pc}/{pd} {
      allow read: if (hasProj(pid) && pc in ['cl','lists','tasks','log','appts','meta','photos','mat','dmg','x']) || (isBot() && pc == 'photos');
      allow create: if okw(pid, pc) || (pc == 'photos' && (dfPh(pd, pid) || twPh(pd, pid)));
      // גרסה 422: פרוטוקול מסירה חתום (x/hoS_*) — רק הבעלים משנה או מוחק. אחרי שהבעלים פתח אותו אפשר לחתום מחדש
      allow update: if okw(pid, pc) && (!(pc == 'x' && pd.matches('hoS_.*')) || isOwner() || resource.data.get('del', 0) == 1);
      allow delete: if isOwner() && pc in ['cl','lists','tasks','log','appts','meta','photos','mat','dmg','x'];
    }
    // סאן מרטין 3 (הנתונים הקיימים, בלי שינוי מיקום)
    match /{col}/{d} {
      allow read: if (hasProj('sm3') && col in ['cl','lists','tasks','log','appts','meta','photos','mat','dmg','x']) || (isBot() && col == 'photos');
      allow create: if okw('sm3', col) || (col == 'photos' && (dfPh(d, 'sm3') || twPh(d, 'sm3')));
      allow update: if okw('sm3', col) && (!(col == 'x' && d.matches('hoS_.*')) || isOwner() || resource.data.get('del', 0) == 1);
      allow delete: if isOwner() && col in ['cl','lists','tasks','log','appts','meta','photos','mat','dmg','x'];
    }
  }
}
